CREATE TABLE IF NOT EXISTS `checkout_cotizaciones` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `checkout_id` BIGINT UNSIGNED NOT NULL,
  `numero_cotizacion` VARCHAR(100) NOT NULL,
  `fecha` DATE NULL,
  `cliente_nombre` VARCHAR(255) NULL,
  `contacto` VARCHAR(255) NULL,
  `email` VARCHAR(255) NULL,
  `telefono` VARCHAR(100) NULL,
  `departamento` VARCHAR(100) NULL,
  `subtotal` INT NOT NULL DEFAULT 0,
  `iva` INT NOT NULL DEFAULT 0,
  `total` INT NOT NULL DEFAULT 0,
  `observaciones` TEXT NULL,
  `estado` ENUM('pendiente', 'autorizada') NOT NULL DEFAULT 'pendiente',
  `created_at` TIMESTAMP NULL DEFAULT NULL,
  `updated_at` TIMESTAMP NULL DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `checkout_cotizaciones_checkout_id_foreign` (`checkout_id`),
  CONSTRAINT `checkout_cotizaciones_checkout_id_foreign` FOREIGN KEY (`checkout_id`) REFERENCES `checkouts` (`id`) ON DELETE CASCADE
) ENGINE=INNODB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS `checkout_cotizacion_detalles` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `checkout_cotizacion_id` BIGINT UNSIGNED NOT NULL,
  `detalle_servicio` TEXT NOT NULL,
  `valor_unitario` INT NOT NULL DEFAULT 0,
  `unidades` INT NOT NULL DEFAULT 1,
  `total_linea` INT NOT NULL DEFAULT 0,
  `created_at` TIMESTAMP NULL DEFAULT NULL,
  `updated_at` TIMESTAMP NULL DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `cotizacion_detalles_cotizacion_id_foreign` (`checkout_cotizacion_id`),
  CONSTRAINT `cotizacion_detalles_cotizacion_id_foreign` FOREIGN KEY (`checkout_cotizacion_id`) REFERENCES `checkout_cotizaciones` (`id`) ON DELETE CASCADE
) ENGINE=INNODB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
